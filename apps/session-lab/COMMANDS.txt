# session-lab — command matrix
#
# Every way to run an experiment or a test, with what each one answers.
# Path is relative to the repository root.
#
#   workspace: /home/dev/Documentos/git_repos/experiments
#   code:      apps/session-lab
#   artefacts: docs/session-lab-runs/  (gitignored)
#
# ─────────────────────────────────────────────────────────────────────────────
# 0. UNIT TESTS — no cluster, seconds
# ─────────────────────────────────────────────────────────────────────────────
cd apps/session-lab && go test ./...
    Answers: does the code still hold together (14 packages).
    Includes: watch options, boot-permit serialisation, hold result, retries,
              store migration, report invariants, riot URL list.

cd apps/session-lab && go vet ./... && go vet -tags integration ./...
    Answers: anything the compiler's static checks can see, integration included.

bash scripts/test-session-lab-experiment.sh
    Answers: does the operator script still honour its contract (help, usage
             errors, --layer mapping, no empty arguments).

# ─────────────────────────────────────────────────────────────────────────────
# 1. THE SCRIPT — the ordinary workload, no real pages
# ─────────────────────────────────────────────────────────────────────────────
scripts/session-lab-experiment.sh --layer session --duration 60s --rate 0.5 --hold 30s
    Answers: does the session layer work at all; per-session cost of Chromium
             on an empty page (about:blank).

scripts/session-lab-experiment.sh --layer infra --duration 60s --rate 0.5 --hold 30s
    Answers: claims and pod scheduling, one session per pod, no browser contexts.

scripts/session-lab-experiment.sh --layer session --duration 300s --rate 0.5 --hold 30s
    Answers: a five-minute stream; steady-state latency and throughput.

SESSION_LAB_LOG_LEVEL=debug scripts/session-lab-experiment.sh --layer session \
  --duration 60s --rate 0.5 --hold 30s
    Answers: the same, with per-session lines (slot, context open/close, cost).

# ─────────────────────────────────────────────────────────────────────────────
# 2. RIOT — real pages (test-only, apps/session-lab/test/)
#    One full League champion page plus a distinct splash per session.
#    This is the only workload whose CPU numbers describe real browsing.
# ─────────────────────────────────────────────────────────────────────────────
cd apps/session-lab

SESSION_LAB_INTEGRATION=1 go test -tags integration -count=1 -timeout 900s \
  ./test/ -run TestRiotLive -v
    Defaults: 18 sessions, 20s holds, 2/s, 1 heavy page. ~30 s.
    Answers: do real pages render, each in its own context.

SESSION_LAB_INTEGRATION=1 RIOT_SESSIONS=300 RIOT_HOLD=75s RIOT_RATE=0.5 \
  go test -tags integration -count=1 -timeout 1800s ./test/ -run TestRiotLive -v
    Ten-minute window, 37.5 offered. Answers: sustained real-page cost.

SESSION_LAB_INTEGRATION=1 RIOT_SESSIONS=300 RIOT_HOLD=30s RIOT_RATE=1 \
  go test -tags integration -count=1 -timeout 1800s ./test/ -run TestRiotLive -v
    300 s window, 30 offered. Answers: the same workload at double the rate.

# MAXIMUM SESSIONS with splash pages only (RIOT_HEAVY=0). Concurrency = rate x hold.
#   rate 1  -> 75 concurrent       rate 2  -> 150       rate 3  -> 225
#   rate 4  -> 300  (~38 browsers at 8 contexts, the physical memory ceiling)
SESSION_LAB_INTEGRATION=1 RIOT_SESSIONS=600 RIOT_HOLD=75s RIOT_RATE=2 RIOT_HEAVY=0 \
  go test -tags integration -count=1 -timeout 2400s ./test/ -run TestRiotLive -v

SESSION_LAB_INTEGRATION=1 RIOT_SESSIONS=30 RIOT_HOLD=30s RIOT_RATE=2 RIOT_HEAVY=10 \
  go test -tags integration -count=1 -timeout 900s ./test/ -run TestRiotLive -v
    Answers: spike convergence — ten full SPAs starting close together.

SESSION_LAB_INTEGRATION=1 RIOT_SESSIONS=60 RIOT_HOLD=30s RIOT_RATE=2 RIOT_BOOT=2 \
  go test -tags integration -count=1 -timeout 900s ./test/ -run TestRiotLive -v
    Answers: does capping simultaneous boots (--boot-concurrency) keep the node
             out of saturation while the same load runs.
